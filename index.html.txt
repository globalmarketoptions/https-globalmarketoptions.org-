  <script>
    (() => {
      const menuButton = document.getElementById("mobile-menu-button");
      const drawer = document.getElementById("mobile-nav");
      const backdrop = document.getElementById("mobile-drawer-backdrop");

      function setDrawerOpen(open) {
        drawer.classList.toggle("is-open", open);
        backdrop.hidden = !open;
        document.body.classList.toggle("drawer-open", open);
        drawer.setAttribute("aria-hidden", String(!open));
        drawer.inert = !open;
        menuButton.setAttribute("aria-expanded", String(open));
        menuButton.setAttribute("aria-label",
          open ? "Close navigation menu" : "Open navigation menu");

        if (open) {
          const firstLink = drawer.querySelector("a");
          firstLink?.focus();
        } else {
          menuButton.focus();
        }
      }

      // Exposed globally for the inline hamburger and close-button handlers.
      window.toggleMobileDrawer = function(event, forceOpen) {
        if (event) event.preventDefault();
        const currentlyOpen = drawer.classList.contains("is-open");
        setDrawerOpen(typeof forceOpen === "boolean" ? forceOpen : !currentlyOpen);
        return false;
      };

      document.addEventListener("keydown", event => {
        if (event.key === "Escape" && drawer.classList.contains("is-open")) {
          setDrawerOpen(false);
        }
      });

      drawer.querySelectorAll("a").forEach(link => {
        link.addEventListener("click", () => setDrawerOpen(false));
      });

      // Keep both language selectors synchronized. Connect these values to a
      // translation system if the site has one; selection alone does not translate content.
      const languageSelects = [
        document.getElementById("language-select"),
        document.getElementById("drawer-language-select")
      ];
      languageSelects.forEach(select => {
        select.addEventListener("change", () => {
          languageSelects.forEach(other => {
            if (other !== select) other.value = select.value;
          });
          document.documentElement.lang = select.value;
          try {
            localStorage.setItem("site-language", select.value);
          } catch (_) {}
          window.dispatchEvent(new CustomEvent("site:language-change", {
            detail: { language: select.value }
          }));
        });
      });

      try {
        const savedLanguage = localStorage.getItem("site-language");
        if (savedLanguage && ["en", "ro", "fr", "es", "de"].includes(savedLanguage)) {
          languageSelects.forEach(select => { select.value = savedLanguage; });
          document.documentElement.lang = savedLanguage;
        }
      } catch (_) {}

      const supportMessage = document.getElementById("support-message");
      const showSupportMessage = text => {
        supportMessage.textContent = text;
        supportMessage.hidden = false;
      };

      document.getElementById("live-chat-button").addEventListener("click", () => {
        // A real chat provider can listen for this event and open its chat panel.
        const event = new CustomEvent("site:open-live-chat", { cancelable: true });
        const handled = !window.dispatchEvent(event);
        if (!handled) {
          showSupportMessage(
            "Live chat is not connected yet. Please use the Support page."
          );
        }
      });

      document.getElementById("whatsapp-button").addEventListener("click", () => {
        // Configure window.SITE_WHATSAPP_NUMBER with a verified international
        // number, digits only, e.g. "40123456789", before enabling this link.
        const number = String(window.SITE_WHATSAPP_NUMBER || "").replace(/\D/g, "");
        if (!number) {
          showSupportMessage(
            "WhatsApp support is not configured yet. Please use the Support page."
          );
          return;
        }
        window.open("https://wa.me/" + number, "_blank", "noopener,noreferrer");
      });
    })();
  </script>

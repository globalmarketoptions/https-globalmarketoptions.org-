  <script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>
  <script src="/supabase-config.js"></script>
</head>
<body class="auth-page">
  <main class="auth-card">
    <a class="brand" href="/">Global Market Options</a>
    <h1>Welcome back</h1>
    <p>Sign in with your email and password.</p>

    <form id="login-form">
      <label>Email
        <input name="email" type="email" autocomplete="email"
               required maxlength="254">
      </label>

      <label>Password
        <input name="password" type="password" autocomplete="current-password"
               required maxlength="128">
      </label>

      <div class="form-actions">
        <button class="text-button" type="button" id="forgot-password">
          Forgot password?
        </button>
      </div>

      <div id="login-message" class="form-message" role="status"
           aria-live="polite"></div>
      <button class="primary-button" type="submit">Sign in</button>
    </form>

    <p class="auth-switch">New member? <a href="/register.html">Create an account</a></p>
    <p><a href="/">Return to home</a></p>
  </main>

  <script>
    (() => {
      const form = document.getElementById("login-form");
      const message = document.getElementById("login-message");
      const submitButton = form.querySelector('[type="submit"]');
      const resetButton = document.getElementById("forgot-password");

      function getSupabaseClient() {
        if (window.supabaseClient) return window.supabaseClient;

        const config = window.SUPABASE_CONFIG || {};
        const url = window.SUPABASE_URL || config.url;
        const key = window.SUPABASE_ANON_KEY || config.anonKey ||
                    window.SUPABASE_PUBLISHABLE_KEY || config.publishableKey;

        if (url && key && window.supabase?.createClient) {
          return window.supabase.createClient(url, key);
        }
        return null;
      }

      function showMessage(text, type = "") {
        message.textContent = text;
        message.className = "form-message" + (type ? " " + type : "");
      }

      form.addEventListener("submit", async event => {
        event.preventDefault();
        showMessage("");

        if (!form.reportValidity()) return;

        const client = getSupabaseClient();
        if (!client) {
          showMessage("Sign-in is not configured. Please contact site support.", "error");
          return;
        }

        const email = String(new FormData(form).get("email") || "")
          .trim().toLowerCase();
        const password = String(new FormData(form).get("password") || "");

        submitButton.disabled = true;
        submitButton.textContent = "Signing in…";

        try {
          const { error } = await client.auth.signInWithPassword({ email, password });
          if (error) throw error;

          // Change this destination if your authenticated dashboard uses another route.
          window.location.assign("/dashboard.html");
        } catch (error) {
          console.error("Sign-in failed:", error?.message || "Unknown error");
          showMessage(
            "We couldn't sign you in with those details. Check your email and password, or reset your password.",
            "error"
          );
        } finally {
          submitButton.disabled = false;
          submitButton.textContent = "Sign in";
        }
      });

      resetButton.addEventListener("click", async () => {
        showMessage("");

        const client = getSupabaseClient();
        if (!client) {
          showMessage("Password reset is not configured. Please contact site support.", "error");
          return;
        }

        const email = String(new FormData(form).get("email") || "")
          .trim().toLowerCase();

        if (!email) {
          showMessage("Enter your email address first, then select Forgot password.", "error");
          form.elements.email.focus();
          return;
        }

        resetButton.disabled = true;
        try {
          const { error } = await client.auth.resetPasswordForEmail(email, {
            redirectTo: window.location.origin + "/reset-password.html"
          });
          if (error) throw error;

          showMessage(
            "If an account exists for that email, a password-reset link will be sent.",
            "success"
          );
        } catch (error) {
          console.error("Password reset failed:", error?.message || "Unknown error");
          showMessage("We couldn't send the reset email. Please try again later.", "error");
        } finally {
          resetButton.disabled = false;
        }
      });
    })();
  </script>
</body>
